3.2 Monitor work-area query (module, after the _shell32.DragFinish block ~398)

    class MONITORINFO(ctypes.Structure):
        """MONITORINFO for GetMonitorInfoW — work area excludes the taskbar."""

        _fields_ = (
            ("cbSize", wintypes.DWORD),
            ("rcMonitor", wintypes.RECT),
            ("rcWork", wintypes.RECT),
            ("dwFlags", wintypes.DWORD),
        )

    _user32.MonitorFromWindow.argtypes = (wintypes.HWND, wintypes.DWORD)
    _user32.MonitorFromWindow.restype = wintypes.HANDLE
    _user32.GetMonitorInfoW.argtypes = (
        wintypes.HANDLE, ctypes.POINTER(MONITORINFO))
    _user32.GetMonitorInfoW.restype = wintypes.BOOL
    _MONITOR_DEFAULTTONEAREST = 2

    def monitor_work_area(widget):
        """Return (left, top, right, bottom) of the widget's monitor work area.

        Failure (destroyed widget, unexpected Win32 result) returns None so
        callers can keep their unclamped geometry rather than guessing.
        """
        try:
            hwnd = int(widget.winfo_id())
        except (tk.TclError, ValueError, TypeError):
            return None
        try:
            hmon = _user32.MonitorFromWindow(hwnd, _MONITOR_DEFAULTTONEAREST)
            info = MONITORINFO()
            info.cbSize = ctypes.sizeof(MONITORINFO)
            if not hmon or not _user32.GetMonitorInfoW(hmon, ctypes.byref(info)):
                return None
        except (OSError, ValueError, OverflowError):
            return None
        work = info.rcWork
        return (work.left, work.top, work.right, work.bottom)
